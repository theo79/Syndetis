cmake_minimum_required(VERSION 3.22)
# Migration-only guards. Application sources and version remain unchanged.
set(IDF_TARGET "esp32" CACHE STRING "Syndetis target")
if(NOT IDF_TARGET STREQUAL "esp32")
    message(FATAL_ERROR "Syndetis5C requires ESP32")
endif()
execute_process(COMMAND git -C "$ENV{IDF_PATH}" rev-parse HEAD
    OUTPUT_VARIABLE SYNDETIS_SDK_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE SYNDETIS_SDK_RESULT)
if(NOT SYNDETIS_SDK_RESULT EQUAL 0 OR NOT SYNDETIS_SDK_COMMIT STREQUAL "fff9895c82d744c7237be8847347bdd1b07c6643")
    message(FATAL_ERROR "Use the official ESP-IDF v6.1 Git checkout; see LINUX_SETUP.md")
endif()
if(NOT DEFINED SDKCONFIG)
    set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig" CACHE FILEPATH "Generated local configuration")
endif()
if(NOT EXISTS "${SDKCONFIG}")
    configure_file("${CMAKE_CURRENT_LIST_DIR}/sdkconfig.pinned" "${SDKCONFIG}" COPYONLY)
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
set(PROJECT_VER "0.6.0")
project(Syndetis)
